// 👇 CAMBIA ESTO por tu propia clave secreta antes de implementar.
const SECRET_TOKEN = 'cambia-esto-por-tu-clave-secreta';

const SHEET_SCHEMA = {
  Ingresos: ['id', 'fecha', 'concepto', 'valor', 'tipo', 'frecuencia'],
  Gastos: ['id', 'fecha', 'categoria', 'descripcion', 'valor', 'metodoPago'],
  Obligaciones: ['id', 'nombre', 'valor', 'fechaPago', 'frecuencia', 'prioridad', 'estado'],
  Deudas: ['id', 'entidad', 'tipoDeuda', 'saldoInicial', 'saldoActual', 'tasaEA', 'plazoMeses', 'tipoCuota', 'cuotaMensual'],
  PagosDeudas: ['id', 'deudaId', 'fecha', 'valor', 'interes', 'capital', 'saldoRestante'],
  Inversiones: ['id', 'nombre', 'tipo', 'capitalInvertido', 'fechaInicio', 'fechaFin', 'tasaRentabilidad'],
  Metas: ['id', 'nombre', 'valorObjetivo', 'fechaObjetivo', 'valorAhorrado'],
  Presupuestos: ['id', 'categoria', 'limiteMensual', 'mes'],
  Configuracion: ['id', 'clave', 'valor'],
};

/** Monly pide los datos (GET) al abrir la app o al sincronizar manualmente. */
function doGet(e) {
  if (!checkToken(e)) return jsonResponse({ error: 'unauthorized' });
  const data = {};
  Object.keys(SHEET_SCHEMA).forEach((name) => { data[name] = readSheet(name); });
  return jsonResponse({ data });
}

/** Monly envía (POST) el estado completo cada vez que agregas/editas/borras algo. */
function doPost(e) {
  if (!checkToken(e)) return jsonResponse({ error: 'unauthorized' });
  let body;
  try {
    body = JSON.parse(e.postData.contents);
  } catch (err) {
    return jsonResponse({ error: 'invalid_body' });
  }
  Object.keys(SHEET_SCHEMA).forEach((name) => {
    if (body.data && Array.isArray(body.data[name])) writeSheet(name, body.data[name]);
  });
  return jsonResponse({ ok: true });
}

function checkToken(e) {
  const token = (e && e.parameter && e.parameter.token) || '';
  return token === SECRET_TOKEN;
}

function getOrCreateSheet(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(SHEET_SCHEMA[name]);
  }
  return sheet;
}

function readSheet(name) {
  const sheet = getOrCreateSheet(name);
  const values = sheet.getDataRange().getValues();
  if (values.length < 2) return [];
  const headers = values[0];
  return values.slice(1)
    .map((row) => {
      const obj = {};
      headers.forEach((h, i) => { obj[h] = row[i]; });
      return obj;
    })
    .filter((obj) => obj.id !== '' && obj.id !== undefined);
}

function writeSheet(name, rows) {
  const sheet = getOrCreateSheet(name);
  const headers = SHEET_SCHEMA[name];

  sheet.clearContents();
  sheet.appendRow(headers);

  if (rows.length) {
    const range = sheet.getRange(2, 1, rows.length, headers.length);
    // Forzamos texto plano: si no, Google Sheets auto-convierte las fechas
    // "YYYY-MM-DD" a su propio tipo Date, y al leerlas de vuelta se corren
    // de día según la zona horaria de la hoja — el mismo bug de fechas que
    // ya resolvimos en la app, así que lo evitamos aquí también.
    range.setNumberFormat('@');
    const values = rows.map((row) => headers.map((h) => (row[h] !== undefined && row[h] !== null) ? row[h] : ''));
    range.setValues(values);
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
